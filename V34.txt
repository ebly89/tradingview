find_best_trend(indices, values, is_resistance, current_atr, trend_start, min_slope, max_len) =>
    line best_l = na
    int max_right_score = -1
    int max_total_score = -1
    int max_length = -1
    float tolerance = current_atr * atr_mult_trend
    int size = array.size(indices)
    int start_idx = math.max(0, size - 20)
    if size >= 3
        for i = start_idx to size - 2 by 1
            int p1_idx = array.get(indices, i)
            float p1_val = array.get(values, i)
            for i2 = i + 1 to size - 1 by 1
                int p2_idx = array.get(indices, i2)
                float p2_val = array.get(values, i2)
                if p2_idx - p1_idx > max_len
                    continue
                float slope = (p2_val - p1_val) / (p2_idx - p1_idx)
                float abs_slope = math.abs(slope)
                float min_slope_threshold = current_atr * min_slope
                if abs_slope < min_slope_threshold
                    continue
                bool broken = false
                int total_score = 0
                int right_score = 0
                int loop_start = math.max(p1_idx, bar_index - 300)
                for j = loop_start to bar_index by 2
                    float line_val = p1_val + slope * (j - p1_idx)
                    if (is_resistance and close[bar_index - j] > line_val + tolerance) or (not is_resistance and close[bar_index - j] < line_val - tolerance)
                        broken := true
                        break
                if not broken
                    for k = start_idx to size - 1 by 1
                        int check_idx = array.get(indices, k)
                        if check_idx >= p1_idx
                            float check_val = array.get(values, k)
                            float line_at_k = p1_val + slope * (check_idx - p1_idx)
                            if math.abs(check_val - line_at_k) <= tolerance
                                total_score := total_score + 1
                                if not na(trend_start) and check_idx > trend_start
                                    right_score := right_score + 1
                bool valid_right = na(trend_start) ? true : right_score >= 1
                if total_score >= 3 and valid_right
                    int current_length = p2_idx - p1_idx
                    if right_score > max_right_score or (right_score == max_right_score and total_score > max_total_score) or (right_score == max_right_score and total_score == max_total_score and current_length > max_length)
                        max_right_score := right_score
                        max_total_score := total_score
                        max_length := current_length
                        if not na(best_l)
                            line.delete(best_l)
                        best_l := line.new(p1_idx, p1_val, p2_idx, p2_val, xloc = xloc.bar_index, extend = extend.right, width = 2)
    best_l
find_parallel_line(indices_opp, values_opp, slope, current_atr, is_main_support, main_p1_idx, main_p1_val) =>
    line best_parallel = na
    float tolerance = current_atr * atr_mult_trend
    int size_opp = array.size(indices_opp)
    int max_parallel_score = -1
    int max_parallel_length = -1
    if size_opp >= 2
        int start_idx = math.max(0, size_opp - 20)
        for i = start_idx to size_opp - 2 by 1
            int opp1_idx = array.get(indices_opp, i)
            float opp1_val = array.get(values_opp, i)
            for i2 = i + 1 to size_opp - 1 by 1
                int opp2_idx = array.get(indices_opp, i2)
                float opp2_val = array.get(values_opp, i2)
                float opp_slope = (opp2_val - opp1_val) / (opp2_idx - opp1_idx)
                float slope_diff_tolerance = tolerance / math.max(1, opp2_idx - opp1_idx)
                if math.abs(opp_slope - slope) <= slope_diff_tolerance
                    float parallel_val_at_main_p1 = opp1_val + slope * (main_p1_idx - opp1_idx)
                    bool valid_position = false
                    if is_main_support and parallel_val_at_main_p1 > main_p1_val + tolerance
                        valid_position := true
                    else if not is_main_support and parallel_val_at_main_p1 < main_p1_val - tolerance
                        valid_position := true
                    if valid_position
                        bool broken = false
                        int score = 0
                        int loop_start = math.max(opp1_idx, bar_index - 300)
                        for j = loop_start to bar_index by 2
                            float line_val = opp1_val + slope * (j - opp1_idx)
                            if is_main_support
                                if close[bar_index - j] > line_val + tolerance
                                    broken := true
                                    break
                            else
                                if close[bar_index - j] < line_val - tolerance
                                    broken := true
                                    break
                        if not broken
                            for k = start_idx to size_opp - 1 by 1
                                int check_idx = array.get(indices_opp, k)
                                if check_idx >= opp1_idx
                                    float check_val = array.get(values_opp, k)
                                    float line_at_k = opp1_val + slope * (check_idx - opp1_idx)
                                    if math.abs(check_val - line_at_k) <= tolerance
                                        score := score + 1
                        if score >= 2
                            int current_length = opp2_idx - opp1_idx
                            if score > max_parallel_score or (score == max_parallel_score and current_length > max_parallel_length)
